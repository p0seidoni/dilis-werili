    <script>
        let isOpened = false;

        function openEnvelope() {
            if (isOpened) return; // რომ ორჯერ არ დაეჭიროს
            isOpened = true;

            const flap = document.getElementById('flap');
            const letter = document.getElementById('love-letter');
            const letterText = document.getElementById('letter-text');
            const grass = document.getElementById('grass');

            // 1. კონვერტის თავსახურის ახდა
            flap.style.transform = 'rotateX(180deg)';
            
            setTimeout(() => {
                flap.style.zIndex = '1';
            }, 400); // z-index-ის შეცვლა რომ ფურცელმა ამოიწიოს

            // 2. ფურცლის ამოსვლა და გადიდება
            setTimeout(() => {
                letter.style.height = 'auto';
                letter.style.minHeight = '300px';
                letter.style.transform = 'translateY(-220px) scale(1.1)';
                letter.style.zIndex = '20'; // ფურცელი მოდის წინ
                letterText.style.opacity = '1'; // ტექსტი ჩნდება
            }, 800);

            // 3. ფონის შეცვლა და ბალახების ამოსვლა
            setTimeout(() => {
                document.body.style.backgroundColor = '#ffb6c1'; // ვარდისფერი ფონი
                grass.style.bottom = '-10px'; // ბალახების გამოჩენა
                startHeartRain(); // გულების წვიმის დაწყება
            }, 1500);
        }

        // გულების წვიმის ფუნქცია
        function startHeartRain() {
            const container = document.getElementById('hearts-container');
            const hearts = ['❤️', '💖', '💚', '✨', '💕']; // სიმბოლოები რომლებიც იწვიმებს

            setInterval(() => {
                const heart = document.createElement('div');
                heart.classList.add('falling-heart');
                // შემთხვევითი გულის არჩევა მასივიდან
                heart.innerText = hearts[Math.floor(Math.random() * hearts.length)];
                
                // შემთხვევითი პოზიცია და ზომა
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.animationDuration = Math.random() * 3 + 2 + 's'; // 2-დან 5 წამამდე
                heart.style.fontSize = Math.random() * 1.5 + 1 + 'rem';
                
                container.appendChild(heart);

                // წაშლა მას მერე რაც ეკრანს გასცდება, რათა ბრაუზერი არ გადაიტვირთოს
                setTimeout(() => {
                    heart.remove();
                }, 5000);
            }, 150); // ყოველ 150 მილიწამში ახალი გული ეცემა
        }
    </script>
